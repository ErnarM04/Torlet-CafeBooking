User Management Models The user management system implements a hierarchical structure with a base User model and specialized profiles for different user types. 4.1 Base User Model The User model extends Django's AbstractUser and uses phone number as the primary authentication field instead of username, which is common in Kazakhstan and Central Asian markets. 💡 NOTE: File location: apps/users/models.py from django.contrib.auth.models import AbstractUser from django.db import models import uuid class User(AbstractUser): """ Base user model for authentication Key Features: - Phone number authentication (no username) - UUID primary key for security - Multi-language support (RU/KK/EN) - Email verification """ # Remove default username field username = None # Primary identifier user_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, verbose_name='User ID' ) # Authentication fields phone_number = models.CharField( max_length=20, unique=True, verbose_name='Phone Number', help_text='Format: +7XXXXXXXXXX' ) password = models.CharField(max_length=128) # Personal information email = models.EmailField(blank=True) first_name = models.CharField(max_length=100, verbose_name='First Name') last_name = models.CharField(max_length=100, verbose_name='Last Name') # Preferences language = models.CharField( max_length=2, choices=[ ('ru', 'Russian'), ('kk', 'Kazakh'), ('en', 'English') ], default='ru', verbose_name='Preferred Language' ) # Verification status is_verified = models.BooleanField( default=False, verbose_name='Phone Verified' ) # Timestamps created_at = models.DateTimeField( auto_now_add=True, verbose_name='Created At' ) last_login = models.DateTimeField( null=True, blank=True, verbose_name='Last Login' ) # Django auth configuration USERNAME_FIELD = 'phone_number' REQUIRED_FIELDS = ['email', 'first_name', 'last_name'] class Meta: db_table = 'users' verbose_name = 'User' verbose_name_plural = 'Users' indexes = [ models.Index(fields=['phone_number']), models.Index(fields=['email']), ] def __str__(self): return f"{self.get_full_name()} ({self.phone_number})" def get_full_name(self): return f"{self.first_name} {self.last_name}" 4.2 Customer Profile The Customer model extends User and adds booking-related functionality and preferences. class Customer(models.Model): """ Customer profile with booking history and preferences Extends User with: - Booking statistics - Loyalty points - Preferences (dietary, seating, etc.) """ user = models.OneToOneField( User, on_delete=models.CASCADE, related_name='customer_profile', primary_key=True ) customer_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, verbose_name='Customer ID' ) # Booking statistics total_bookings = models.IntegerField( default=0, verbose_name='Total Bookings' ) cancelled_bookings = models.IntegerField( default=0, verbose_name='Cancelled Bookings' ) no_show_count = models.IntegerField( default=0, verbose_name='No-Show Count' ) # Loyalty program loyalty_points = models.IntegerField( default=0, verbose_name='Loyalty Points' ) # Personal preferences date_of_birth = models.DateField( null=True, blank=True, verbose_name='Date of Birth' ) preferences = models.JSONField( default=dict, blank=True, verbose_name='Preferences', help_text='Store dietary restrictions, seating preferences, etc.' ) # Example preferences structure: # { # "dietary": ["vegetarian", "gluten-free"], # "seating": "window", # "special_occasions": ["birthday_discount"], # "communication": { # "sms": True, # "email": True, # "push": True # } # } class Meta: db_table = 'customers' verbose_name = 'Customer' verbose_name_plural = 'Customers' def __str__(self): return f"Customer: {self.user.get_full_name()}" @property def reliability_score(self): """Calculate customer reliability (0-100)""" if self.total_bookings == 0: return 100 completed = self.total_bookings - self.cancelled_bookings - self.no_show_count score = (completed / self.total_bookings) * 100 return round(score, 2) 💡 NOTE: The reliability_score property helps restaurants prioritize reliable customers and can be used in waitlist management. 4.3 Staff Models Three specialized staff models handle different permission levels: class Administrator(models.Model): """System administrator with full access""" user = models.OneToOneField( User, on_delete=models.CASCADE, related_name='admin_profile', primary_key=True ) admin_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True ) role = models.CharField( max_length=50, choices=[ ('super_admin', 'Super Administrator'), ('admin', 'Administrator'), ('moderator', 'Moderator') ] ) class Meta: db_table = 'administrators' class RestaurantOwner(models.Model): """Restaurant owner who can manage their restaurants""" user = models.OneToOneField( User, on_delete=models.CASCADE, related_name='owner_profile', primary_key=True ) owner_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True ) # Reverse relation: owned_restaurants (from Restaurant model) class Meta: db_table = 'restaurant_owners' class RestaurantStaff(models.Model): """Staff member assigned to a specific restaurant""" user = models.OneToOneField( User, on_delete=models.CASCADE, related_name='staff_profile', primary_key=True ) staff_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True ) restaurant = models.ForeignKey( 'cafes.Restaurant', on_delete=models.CASCADE, related_name='staff' ) position = models.CharField( max_length=50, choices=[ ('waiter', 'Waiter'), ('manager', 'Manager'), ('host', 'Host'), ('chef', 'Chef') ] ) hire_date = models.DateField( verbose_name='Hire Date' ) is_active = models.BooleanField( default=True, verbose_name='Active Employment' ) class Meta: db_table = 'restaurant_staff' verbose_name = 'Restaurant Staff' verbose_name_plural = 'Restaurant Staff' 5. Restaurant & Location Models 5.1 Restaurant Model The Restaurant model represents the main entity. A restaurant can have multiple locations. class Restaurant(models.Model): """ Main restaurant/cafe entity Features: - Multi-location support - Rating aggregation - Cuisine type categorization - Owner management """ restaurant_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, primary_key=True ) # Basic information name = models.CharField( max_length=200, verbose_name='Restaurant Name' ) description = models.TextField( blank=True, verbose_name='Description' ) cuisine_type = models.CharField( max_length=100, verbose_name='Cuisine Type', help_text='e.g., Italian, Asian, Kazakh, International' ) # Primary address (for display) address = models.TextField(verbose_name='Primary Address') city = models.CharField(max_length=100, default='Almaty') # Coordinates for map display latitude = models.DecimalField( max_digits=9, decimal_places=6, null=True, blank=True ) longitude = models.DecimalField( max_digits=9, decimal_places=6, null=True, blank=True ) # Media images = models.JSONField( default=list, blank=True, verbose_name='Image URLs', help_text='List of image URLs' ) # Example: ["https://cdn.example.com/rest1.jpg", "https://cdn.example.com/rest2.jpg"] # Rating & reviews rating = models.FloatField( default=0.0, verbose_name='Average Rating' ) total_reviews = models.IntegerField( default=0, verbose_name='Total Reviews' ) # Ownership owner = models.ForeignKey( 'users.RestaurantOwner', on_delete=models.SET_NULL, null=True, related_name='owned_restaurants' ) # Status is_active = models.BooleanField( default=True, verbose_name='Active' ) # Metadata created_at = models.DateTimeField( auto_now_add=True, verbose_name='Created At' ) updated_at = models.DateTimeField( auto_now=True, verbose_name='Updated At' ) class Meta: db_table = 'restaurants' verbose_name = 'Restaurant' verbose_name_plural = 'Restaurants' indexes = [ models.Index(fields=['city', 'cuisine_type']), models.Index(fields=['rating']), models.Index(fields=['-created_at']), ] def __str__(self): return self.name def calculate_rating(self): """Recalculate average rating from reviews""" from django.db.models import Avg from reviews.models import Review result = Review.objects.filter( restaurant=self, is_approved=True ).aggregate(Avg('rating')) self.rating = result['rating__avg'] or 0.0 self.total_reviews = Review.objects.filter( restaurant=self, is_approved=True ).count() self.save(update_fields=['rating', 'total_reviews']) 5.2 Location Model class Location(models.Model): """ Physical location/branch of a restaurant Why separate from Restaurant: - Restaurants can have multiple branches - Each location has its own: * Operating hours * Tables * Time slots * Capacity """ location_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, primary_key=True ) restaurant = models.ForeignKey( Restaurant, on_delete=models.CASCADE, related_name='locations' ) # Address address = models.TextField(verbose_name='Full Address') city = models.CharField(max_length=100, default='Almaty') # Coordinates latitude = models.DecimalField( max_digits=9, decimal_places=6, null=True ) longitude = models.DecimalField( max_digits=9, decimal_places=6, null=True ) # Operating hours (simplified) opening_hours = models.CharField( max_length=200, verbose_name='Opening Hours', help_text='e.g., Mon-Fri 10:00-22:00, Sat-Sun 11:00-23:00' ) # For more complex hours, use BusinessHours model (separate) # Capacity total_capacity = models.IntegerField( default=0, verbose_name='Total Seating Capacity' ) # Status is_active = models.BooleanField( default=True, verbose_name='Location Active' ) created_at = models.DateTimeField(auto_now_add=True) updated_at = models.DateTimeField(auto_now=True) class Meta: db_table = 'locations' verbose_name = 'Location' verbose_name_plural = 'Locations' indexes = [ models.Index(fields=['restaurant', 'is_active']), models.Index(fields=['city']), ] def __str__(self): return f"{self.restaurant.name} - {self.city}" def update_capacity(self): """Recalculate total capacity from active tables""" from django.db.models import Sum result = self.tables.filter( is_active=True ).aggregate(Sum('max_guests')) self.total_capacity = result['max_guests__sum'] or 0 self.save(update_fields=['total_capacity']) 5.3 Table Model class Table(models.Model): """ Individual table at a location Key concepts: - Table types (indoor, outdoor, VIP, terrace) - Capacity range (min/max guests) - Availability tracking - Position for floor plan display """ table_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, primary_key=True ) location = models.ForeignKey( Location, on_delete=models.CASCADE, related_name='tables' ) # Table identification table_number = models.CharField( max_length=10, verbose_name='Table Number', help_text='e.g., T1, A5, VIP-3' ) # Table type table_type = models.CharField( max_length=20, choices=[ ('indoor', 'Indoor'), ('outdoor', 'Outdoor'), ('vip', 'VIP'), ('terrace', 'Terrace'), ('bar', 'Bar Seating'), ('private', 'Private Room') ], default='indoor' ) # Capacity min_guests = models.IntegerField( default=1, verbose_name='Minimum Guests' ) max_guests = models.IntegerField( verbose_name='Maximum Guests' ) # For backward compatibility with UML capacity = models.IntegerField( editable=False, verbose_name='Capacity (deprecated)' ) # Floor plan position (optional, for UI) position_x = models.IntegerField( null=True, blank=True, verbose_name='X Position on Floor Plan' ) position_y = models.IntegerField( null=True, blank=True, verbose_name='Y Position on Floor Plan' ) # Description description = models.TextField( blank=True, help_text='Special features, view, etc.' ) # Status is_available = models.BooleanField( default=True, verbose_name='Currently Available' ) is_active = models.BooleanField( default=True, verbose_name='Active (not under maintenance)' ) class Meta: db_table = 'tables' verbose_name = 'Table' verbose_name_plural = 'Tables' unique_together = ['location', 'table_number'] indexes = [ models.Index(fields=['location', 'is_active', 'is_available']), models.Index(fields=['table_type']), ] def __str__(self): return f"{self.location.restaurant.name} - Table {self.table_number}" def save(self, *args, **kwargs): # Set capacity to max_guests for backward compatibility self.capacity = self.max_guests super().save(*args, **kwargs) def check_capacity(self, number_of_guests): """Check if table can accommodate the number of guests""" return self.min_guests <= number_of_guests <= self.max_guests 5.4 TimeSlot Model class TimeSlot(models.Model): """ Available time slots for booking at a location Use cases: - Define booking windows (e.g., 12:00-14:00 for lunch) - Set different slots for weekdays vs weekends - Control booking density """ time_slot_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, primary_key=True ) location = models.ForeignKey( Location, on_delete=models.CASCADE, related_name='time_slots' ) # Time window start_time = models.TimeField(verbose_name='Start Time') end_time = models.TimeField(verbose_name='End Time') # Duration (calculated or explicit) duration = models.CharField( max_length=50, verbose_name='Duration', help_text='e.g., "2 hours", "90 minutes"' ) # Days of week this slot is active days_of_week = models.JSONField( default=list, verbose_name='Active Days', help_text='List of day names: ["Monday", "Tuesday", ...]' ) # Example: ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday"] # Maximum concurrent bookings (optional capacity control) max_bookings = models.IntegerField( default=10, verbose_name='Maximum Bookings per Slot' ) # Status is_active = models.BooleanField( default=True, verbose_name='Active' ) class Meta: db_table = 'time_slots' verbose_name = 'Time Slot' verbose_name_plural = 'Time Slots' ordering = ['start_time'] indexes = [ models.Index(fields=['location', 'is_active']), ] def __str__(self): return f"{self.start_time} - {self.end_time}" @property def duration_minutes(self): """Calculate duration in minutes""" from datetime import datetime, timedelta start = datetime.combine(datetime.today(), self.start_time) end = datetime.combine(datetime.today(), self.end_time) if end < start: # Crosses midnight end += timedelta(days=1) delta = end - start return int(delta.total_seconds() / 60) 6. Booking System Models The booking system is the core of the application. It handles reservations, conflict detection, status management, and waitlist coordination. 6.1 Booking Model class Booking(models.Model): """ Main booking entity Workflow: 1. Customer creates booking (status: pending) 2. System checks availability 3. Table is assigned (status: confirmed) 4. Customer arrives (status: seated) 5. Service completed (status: completed) Alternative flows: - Customer cancels (status: cancelled) - Customer doesn't show up (status: no_show) """ booking_id = models.UUIDField( default=uuid.uuid4, editable=False, unique=True, primary_key=True ) # Human-readable booking number booking_number = models.CharField( max_length=20, unique=True, editable=False, verbose_name='Booking Number' ) # Format: BK-20260203-12345 # Relations customer = models.ForeignKey( 'users.Customer', on_delete=models.CASCADE, related_name='bookings' ) restaurant = models.ForeignKey( 'cafes.Restaurant', on_delete=models.CASCADE, related_name='bookings' ) location = models.ForeignKey( 'cafes.Location', on_delete=models.CASCADE, related_name='bookings' ) table = models.ForeignKey( 'cafes.Table', on_delete=models.SET_NULL, null=True, blank=True, related_name='bookings', help_text='Assigned table (can be null initially)' ) # Booking details booking_date = models.DateField(verbose_name='Date') booking_time = models.TimeField(verbose_name='Time') number_of_guests = models.IntegerField(verbose_name='Number of Guests') duration_minutes = models.IntegerField( default=120, verbose_name='Duration (minutes)' ) # Status tracking STATUS_CHOICES = [ ('pending', 'Pending'), ('confirmed', 'Confirmed'), ('seated', 'Seated'), ('completed', 'Completed'), ('cancelled', 'Cancelled'), ('no_show', 'No Show') ] status = models.CharField( max_length=20, choices=STATUS_CHOICES, default='pending' ) # Additional information special_request = models.TextField( blank=True, verbose_name='Special Requests', help_text='Dietary restrictions, occasion, etc.' ) # Timestamps for status changes created_at = models.DateTimeField(auto_now_add=True) confirmed_at = models.DateTimeField(null=True, blank=True) cancelled_at = models.DateTimeField(null=True, blank=True) completed_at = models.DateTimeField(null=True, blank=True) updated_at = models.DateTimeField(auto_now=True) # Cancellation info cancellation_reason = models.TextField(blank=True) # Staff who handled the booking assigned_by = models.ForeignKey( 'users.RestaurantStaff', on_delete=models.SET_NULL, null=True, blank=True, related_name='assigned_bookings' ) class Meta: db_table = 'bookings' verbose_name = 'Booking' verbose_name_plural = 'Bookings' ordering = ['-booking_date', '-booking_time'] indexes = [ models.Index(fields=['booking_date', 'booking_time']), models.Index(fields=['customer', 'status']), models.Index(fields=['location', 'booking_date']), models.Index(fields=['status']), ] def save(self, *args, **kwargs): if not self.booking_number: self.booking_number = self.generate_booking_number() super().save(*args, **kwargs) def generate_booking_number(self): """Generate unique booking number: BK-YYYYMMDD-XXXXX""" from datetime import datetime import random date_str = datetime.now().strftime('%Y%m%d') random_str = ''.join([str(random.randint(0, 9)) for _ in range(5)]) return f"BK-{date_str}-{random_str}" @property def end_time(self): """Calculate booking end time""" from datetime import datetime, timedelta start = datetime.combine(self.booking_date, self.booking_time) end = start + timedelta(minutes=self.duration_minutes) return end.time() def __str__(self): return f"{self.booking_number} - {self.customer.user.get_full_name()}" 9. Testing Strategy A comprehensive testing strategy ensures code quality and prevents regressions. 11.1 Unit Tests # File: apps/bookings/tests/test_services.py from django.test import TestCase from django.utils import timezone from datetime import datetime, timedelta from apps.bookings.services import BookingService from apps.cafes.models import Restaurant, Location, Table class BookingServiceTest(TestCase): def setUp(self): """Set up test data""" self.restaurant = Restaurant.objects.create( name='Test Restaurant', cuisine_type='Italian', city='Almaty' ) self.location = Location.objects.create( restaurant=self.restaurant, address='Test Address', city='Almaty', opening_hours='10:00-22:00' ) self.table = Table.objects.create( location=self.location, table_number='T1', min_guests=2, max_guests=4 ) def test_check_availability_empty_slot(self): """Test availability check for empty time slot""" date = datetime.today().date() + timedelta(days=1) time = datetime.strptime('18:00', '%H:%M').time() is_available, message = BookingService.check_table_availability( table=self.table, booking_date=date, booking_time=time, duration_minutes=120 ) self.assertTrue(is_available) self.assertEqual(message, 'Table is available') def test_check_availability_conflict(self): """Test availability check with existing booking""" # Create existing booking date = datetime.today().date() + timedelta(days=1) time1 = datetime.strptime('18:00', '%H:%M').time() Booking.objects.create( customer=self.customer, restaurant=self.restaurant, location=self.location, table=self.table, booking_date=date, booking_time=time1, number_of_guests=2, duration_minutes=120, status='confirmed' ) # Try to book overlapping time time2 = datetime.strptime('19:00', '%H:%M').time() is_available, message = BookingService.check_table_availability( table=self.table, booking_date=date, booking_time=time2, duration_minutes=120 ) self.assertFalse(is_available) self.assertIn('already booked', message) 11.2 Running Tests #
